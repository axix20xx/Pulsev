function awgRequest(PDO $pdo, string $method, string $endpoint, array $body = [])
{
    try {
        // Cache settings statically so we only hit the DB once per process execution
        static $cfg = null;
        if ($cfg === null) {
            $cfg = $pdo->query("
                SELECT awg_api_url, awg_api_token
                FROM settings
                LIMIT 1
            ")->fetch(PDO::FETCH_ASSOC);
        }

        if (!$cfg) {
            return null;
        }

        $url = rtrim($cfg['awg_api_url'], '/');
        $ch = curl_init();

        curl_setopt_array($ch, [
            CURLOPT_URL => $url . $endpoint,
            CURLOPT_RETURNTRANSFER => true,
            CURLOPT_TIMEOUT => 15,
            CURLOPT_CONNECTTIMEOUT => 5, // Added connect timeout
            CURLOPT_HTTPHEADER => [
                'Authorization: Bearer ' . $cfg['awg_api_token'],
                'Content-Type: application/json'
            ]
        ]);

        switch ($method) {
            case 'POST':
                curl_setopt($ch, CURLOPT_POST, true);
                curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode($body));
                break;
            case 'PATCH':
                curl_setopt($ch, CURLOPT_CUSTOMREQUEST, 'PATCH');
                curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode($body));
                break;
            case 'DELETE':
                curl_setopt($ch, CURLOPT_CUSTOMREQUEST, 'DELETE');
                break;
        }

        $response = curl_exec($ch);
        $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
        curl_close($ch);

        if ($httpCode >= 300) {
            return null;
        }

        return $response;

    } catch (Throwable $e) {
        return null;
    }
}

function send_telegram_alert($message) {
    global $TG_BOT_TOKEN, $TG_MANAGER_ID;
    if (empty($TG_BOT_TOKEN) || empty($TG_MANAGER_ID)) return;

    $url = "https://api.telegram.org/bot$TG_BOT_TOKEN/sendMessage";
    $post_data = [
        'chat_id' => $TG_MANAGER_ID,
        'text' => $message,
        'parse_mode' => 'HTML', 
        'disable_web_page_preview' => true
    ];

    $ch = curl_init();
    curl_setopt($ch, CURLOPT_URL, $url);
    curl_setopt($ch, CURLOPT_POST, 1);
    curl_setopt($ch, CURLOPT_POSTFIELDS, $post_data);
    curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
    curl_setopt($ch, CURLOPT_TIMEOUT, 5);
    curl_setopt($ch, CURLOPT_CONNECTTIMEOUT, 3); // Added connect timeout
    curl_exec($ch);
    curl_close($ch);
}

function get_db_connection($force_ping = false) {
    global $pdo;
    try {
        if ($pdo) {
            if ($force_ping) {
                $pdo->query("SELECT 1"); 
            }
            return $pdo;
        }
    } catch (Throwable $e) {
        $pdo = null; 
    }

    try {
        if (!$pdo) {
            $pdo = new PDO(DB_DSN, DB_USER, DB_PASS, [
                PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
                PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
                PDO::ATTR_TIMEOUT => 10,
                PDO::ATTR_EMULATE_PREPARES => false, 
            ]);
        }
        return $pdo;
    } catch (Throwable $e) {
        die("DB Error");
    }
}

    $sched_stmt = $pdo_sched->query("
        SELECT s.id as service_id, s.rain_username as username, 
               s.next_plan_gb, s.next_plan_days, s.expire_at, 
               s.gb, s.days,
               COALESCE(c.used_bytes, s.last_used_bytes, 0) as used_bytes
        FROM reseller_services s
        LEFT JOIN reseller_sub_tokens t ON s.id = t.service_id
LEFT JOIN user_stats_cache c ON t.token COLLATE utf8mb4_unicode_ci = c.token COLLATE utf8mb4_unicode_ci
        WHERE s.auto_renew = 1
    ");

    $scheduled_users = $sched_stmt->fetchAll();
    
    foreach ($scheduled_users as $s_user) {
        $used = (float)$s_user['used_bytes'];
        $plan_gb = (float)$s_user['gb'];
        
        $total_plan_bytes = $plan_gb * 1073741824;
        $usage_percent = ($total_plan_bytes > 0) ? ($used / $total_plan_bytes) : 0;
        $near_limit = ($usage_percent >= 0.99);
        
        $expire_ts = strtotime($s_user['expire_at']);
        $time_left = $expire_ts - time();
        $near_expire = ($expire_ts > 0 && $time_left <= 86400);
        
        if ($near_limit || $near_expire) {
            $next_gb = (float)$s_user['next_plan_gb'];
            if ($next_gb <= 0) $next_gb = $plan_gb;
            
            $next_days = (int)$s_user['next_plan_days'];
            if ($next_days <= 0) $next_days = (int)$s_user['days'];
            
            if ($next_gb > 0 && $next_days > 0) {
                $payload = json_encode(['gb' => $next_gb, 'days' => $next_days, 'enable' => true]);
                
                $q_stmt = $pdo_sched->prepare("INSERT INTO reseller_sync_queue (service_id, username, action, payload, status, created_at, updated_at) VALUES (?, ?, 'renew', ?, 'pending', NOW(), NOW())");
                $q_stmt->execute([$s_user['service_id'], $s_user['username'], $payload]);
                
                $u_stmt = $pdo_sched->prepare("UPDATE reseller_services SET auto_renew = 0, next_plan_gb = NULL, next_plan_days = NULL WHERE id = ?");
                $u_stmt->execute([$s_user['service_id']]);

                send_telegram_alert("♻️ <b>تمدید خودکار</b>\nکاربر <code>{$s_user['username']}</code> به دلیل پایان سرویس (یا حجم) بصورت خودکار تمدید شد.\nحجم جدید: {$next_gb}GB\nزمان جدید: {$next_days} روز");
            }
        }
    }
} catch (Throwable $e) {
    error_log("Early Renew Logic Error: " . $e->getMessage());

                    $error_count = count($current_errors);
                    $msg .= "\n🛑 <b>خطاها ($error_count):</b>\n";
                    $shown_errors = array_slice($current_errors, 0, 3);
                    foreach($shown_errors as $err) {
                        $msg .= "▪️ <pre>" . htmlspecialchars($err) . "</pre>\n";
                    }
                }
                $msg .= "➖➖➖➖➖➖➖➖➖➖\n";
                $msg .= "🕒 " . date('Y-m-d H:i:s');

                send_telegram_alert($msg);
            } 
        }
    } catch (Throwable $e) {
        send_telegram_alert("🔥 <b>CRITICAL ERROR</b>\nJob ID: $jobId\nError: " . htmlspecialchars($e->getMessage()));
        
        try {
            $crash_pdo = get_db_connection(true); 
            $crash_log = json_encode(['log' => "CRASH: " . $e->getMessage(), 'done' => $done_panels_list ?? []]);
            $crash_stmt = $crash_pdo->prepare("UPDATE reseller_sync_queue SET status='failed', last_error=?, attempts=attempts+1 WHERE id=?");
            $crash_stmt->execute([$crash_log, $jobId]);
        } catch (Throwable $ex) { /* Ignore */ }
    }
}
