function detect_real_sni(array $ob): string {

    // 1️⃣ explicit TLS server_name
    if (!empty($ob['tls']['server_name'])) {
        return is_array($ob['tls']['server_name'])
            ? $ob['tls']['server_name'][0]
            : $ob['tls']['server_name'];
    }

    // 2️⃣ WS Host header
    if (!empty($ob['transport']['headers']['Host'])) {
        return is_array($ob['transport']['headers']['Host'])
            ? $ob['transport']['headers']['Host'][0]
            : $ob['transport']['headers']['Host'];
    }

    // 3️⃣ XHTTP host
    if (!empty($ob['transport']['host'])) {
        return is_array($ob['transport']['host'])
            ? $ob['transport']['host'][0]
            : $ob['transport']['host'];
    }

    // 4️⃣ TCP HTTP host
    if (!empty($ob['transport']['header']['request']['headers']['Host'])) {
        return is_array($ob['transport']['header']['request']['headers']['Host'])
            ? $ob['transport']['header']['request']['headers']['Host'][0]
            : $ob['transport']['header']['request']['headers']['Host'];
    }

    // 🚫 DO NOT fallback to server
    return '';
}

function fetch_stats_headers($url) {
    // Basic validation
    if (!filter_var($url, FILTER_VALIDATE_URL)) return null;

    $mh = curl_multi_init();

    // Common options for maximum compatibility
    $commonOptions = [
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_HEADER         => true,
        CURLOPT_FOLLOWLOCATION => true,
        CURLOPT_SSL_VERIFYHOST => 0,
        CURLOPT_SSL_VERIFYPEER => 0,
        CURLOPT_USERAGENT      => 'v2rayNG',
        // INCREASED TIMEOUTS HERE
        CURLOPT_CONNECTTIMEOUT => 4,   // Increased to 10s for slow handshakes
        CURLOPT_TIMEOUT        => 15,   // Increased to 15s for total operation time
    ];

    // --- Request 1: HEAD (Lightweight) ---
    $ch_head = curl_init($url);
    curl_setopt_array($ch_head, $commonOptions);
    curl_setopt($ch_head, CURLOPT_NOBODY, true);
    curl_multi_add_handle($mh, $ch_head);

    // --- Request 2: GET (Fallback with Range) ---
    $ch_get = curl_init($url);
    curl_setopt_array($ch_get, $commonOptions);
    curl_setopt($ch_get, CURLOPT_NOBODY, false);
    curl_setopt($ch_get, CURLOPT_RANGE, '0-10');
    curl_multi_add_handle($mh, $ch_get);

    $active = null;
    $finalData = null;

    // Start execution
    do {
        $status = curl_multi_exec($mh, $active);

        // Check for completed requests
        while ($info = curl_multi_info_read($mh)) {
            $handle = $info['handle'];
            $content = curl_multi_getcontent($handle);

            // Check if this specific request succeeded and has the header
            if ($content && stripos($content, 'Subscription-Userinfo') !== false) {
                if (preg_match('/^Subscription-Userinfo:\s*(.*)$/im', $content, $matches)) {
                    $data = [];
                    $parts = explode(';', $matches[1]);
                    foreach ($parts as $part) {
                        $kv = explode('=', trim($part));
                        if (count($kv) === 2) {
                            $data[trim($kv[0])] = (float)trim($kv[1]);
                        }
                    }
                    
                    if (!empty($data)) {
                        $finalData = $data;
                        break 2;
                    }
                }
            }
            curl_multi_remove_handle($mh, $handle);
        }

        if ($active) {
            if (curl_multi_select($mh) == -1) {
                usleep(100000);
            }
        }

    } while ($active && $status == CURLM_OK);

    curl_multi_remove_handle($mh, $ch_head);
    curl_multi_remove_handle($mh, $ch_get);
    curl_multi_close($mh);

    return $finalData;
}

function normalize_azadpanel_sub_url(string $sub, array $panel): ?string {
    $sub = trim($sub);
    if ($sub === '') return null;

    // 1. If the URL is already absolute (starts with http/https), return it as is.
    if (preg_match('/^https?:\/\//i', $sub)) {
        return $sub;
    }

    // 2. It is a relative path (e.g., /sub/xyz...). We need to attach the panel's domain.
    $apiUrl = $panel['api_base'] ?? $panel['api_url'] ?? '';
    if ($apiUrl === '') return $sub; // Can't fix it without a base URL

    // 3. Parse the API URL to extract just the Root Domain (Scheme + Host + Port)
    // This removes paths like '/api/v1' which cause 404s when combined with '/sub/'
    $parsed = parse_url($apiUrl);
    if (!$parsed || !isset($parsed['host'])) {
        // Fallback: If parsing fails, just prepend the full API URL
        return rtrim($apiUrl, '/') . '/' . ltrim($sub, '/');
    }

    $scheme = $parsed['scheme'] ?? 'https';
    $host   = $parsed['host'];
    $port   = isset($parsed['port']) ? ':' . $parsed['port'] : '';
    
    $baseDomain = "$scheme://$host$port";

    // 4. Return the clean URL: Domain + Sub Path
    return $baseDomain . '/' . ltrim($sub, '/');
}

function rename_node_simple(array $node, int $index, string $prefix = '', string $remainText = ''): array {
    $emoji = '';
    if (isset($node['tag'])) {
        if (preg_match_all('/([\x{1F1E6}-\x{1F1FF}]{2})/u', $node['tag'], $matches)) {
            foreach ($matches[0] as $flag) {
                if ($flag === '🇮🇷') {
                    $emoji = '🛜';
                    break;
                }
                $emoji = $flag;
            }
        }
    }

    // Construct the new tag: Server 1 - SPE - 📊 15.5GB 🛜
    $tagParts = ["Server $index"];
    if ($prefix !== '') $tagParts[] = "- $prefix";
    if ($remainText !== '') $tagParts[] = "- 📊 $remainText";
    if ($emoji !== '') $tagParts[] = $emoji;
    
    $node['tag'] = trim(implode(' ', $tagParts));
    return $node;
}

function singbox_node_to_clash(array $n): ?array {

    if (empty($n['type']) || empty($n['server']) || empty($n['server_port'])) {
        return null;
    }

    $p = [
'name' => !empty($n['tag'])
    ? $n['tag']
    : ($n['server'] . ':' . $n['server_port']),
        'server' => $n['server'],
        'port'   => (int)$n['server_port'],
        'udp'    => true,
    ];

    $tls = $n['tls'] ?? [];
    $tlsEnabled = !empty($tls['enabled']);

    // ---------- SAFE SNI DETECTION ----------
    $servername='';

    if (!empty($tls['server_name'])) {
        $servername=is_array($tls['server_name'])?reset($tls['server_name']):$tls['server_name'];
    }
    elseif (!empty($n['sni'])) {
        $servername=$n['sni'];
    }
    elseif (!empty($n['transport']['host'])) {
        $servername=is_array($n['transport']['host'])?reset($n['transport']['host']):$n['transport']['host'];
    }
    elseif (!empty($n['transport']['headers']['Host'])) {
        $servername=$n['transport']['headers']['Host'];
    }

    // ---------- HELPERS ----------
    $normalizeAlpn=function($a){
        if(empty($a)) return null;
        if(is_array($a)) return array_values(array_filter($a));
        return array_values(array_filter(array_map('trim',explode(',',$a))));
    };

    $normalizeHeaders=function($h){
        if(!is_array($h)) return [];
        $o=[];
        foreach($h as $k=>$v){
            if($v===''||$v===null) continue;
            $o[$k]=is_array($v)?reset($v):$v;
        }
        return $o;
    };

    $ensureHost=function(&$headers,$servername){
        if($servername!=='' && empty($headers['Host'])) $headers['Host']=$servername;
    };

    $applyTLS=function(&$p,$tls,$tlsEnabled,$servername,$normalizeAlpn){
        $p['tls']=$tlsEnabled;

        if($servername!=='') $p['servername']=$servername;

        if(!empty($tls['utls']['fingerprint'])){
            $p['client-fingerprint']=$tls['utls']['fingerprint'];
        }

        if(!empty($tls['insecure'])) $p['skip-cert-verify']=true;

        $alpn=$normalizeAlpn($tls['alpn']??null);
        if(!$alpn && $tlsEnabled) $alpn=['h2','http/1.1'];
        if($alpn) $p['alpn']=$alpn;

        // REALITY FULL SUPPORT
        if(!empty($tls['reality']['enabled'])){
            $p['reality-opts']=[
                'public-key'=>$tls['reality']['public_key']??'',
                'short-id'=>$tls['reality']['short_id']??'',
            ];
            if(!empty($tls['reality']['spider_x']))
                $p['reality-opts']['spider-x']=$tls['reality']['spider_x'];
        }
    };

    // ================= TYPES =================

    switch(strtolower($n['type'])){

        case 'vless':
            $p['type']='vless';
            $p['uuid']=$n['uuid']??'';
            if(!empty($n['flow'])) $p['flow']=$n['flow'];
            $applyTLS($p,$tls,$tlsEnabled,$servername,$normalizeAlpn);
        break;

        case 'vmess':
            $p['type']='vmess';
            $p['uuid']=$n['uuid']??'';
            $p['alterId']=0;
            $p['cipher']='auto';
            $applyTLS($p,$tls,$tlsEnabled,$servername,$normalizeAlpn);
        break;

        case 'trojan':
            $p['type']='trojan';
            $p['password']=$n['password']??'';
            $applyTLS($p,$tls,true,$servername,$normalizeAlpn);
        break;

        case 'shadowsocks':
        case 'ss':
            $p['type']='ss';
            $p['cipher']=$n['method']??'aes-256-gcm';
            $p['password']=$n['password']??'';
        break;

        // ---------- HYSTERIA2 ----------
        case 'hysteria2':
        case 'hy2':
            $p['type']='hysteria2';
            $p['password']=$n['password']??'';
            if(!empty($n['up_mbps'])) $p['up']=$n['up_mbps'];
            if(!empty($n['down_mbps'])) $p['down']=$n['down_mbps'];
            if($servername!=='') $p['sni']=$servername;
        break;

        // ---------- TUIC ----------
        case 'tuic':
            $p['type']='tuic';
            $p['uuid']=$n['uuid']??'';
            $p['password']=$n['password']??'';
            if(!empty($n['congestion_control'])) $p['congestion-controller']=$n['congestion_control'];
            if($servername!=='') $p['sni']=$servername;
        break;

        // ---------- SOCKS ----------
        case 'socks':
            $p['type']='socks5';
            if(!empty($n['username'])) $p['username']=$n['username'];
            if(!empty($n['password'])) $p['password']=$n['password'];
        break;

        // ---------- HTTP PROXY ----------
        case 'http':
            $p['type']='http';
            if(!empty($n['username'])) $p['username']=$n['username'];
            if(!empty($n['password'])) $p['password']=$n['password'];
        break;

        default:
            return null;
    }

    // ================= TRANSPORT =================

    $transport=$n['transport']??[];
    $type=strtolower($transport['type']??'tcp');

    if($type==='ws'){
        $p['network']='ws';
        $headers=$normalizeHeaders($transport['headers']??[]);
        if(!empty($transport['host'])){
            $headers['Host']=is_array($transport['host'])?reset($transport['host']):$transport['host'];
        }
        $ensureHost($headers,$servername);
        $p['ws-opts']=[
            'path'=>$transport['path']??'/',
            'headers'=>$headers,
        ];
    }

    elseif($type==='grpc'){
        $p['network']='grpc';
        $p['grpc-opts']=[
            'grpc-service-name'=>$transport['service_name']??'',
        ];
    }

    elseif($type==='httpupgrade'){
        $p['network']='httpupgrade';
        $headers=$normalizeHeaders($transport['headers']??[]);
        if(!empty($transport['host'])){
            $headers['Host']=is_array($transport['host'])?reset($transport['host']):$transport['host'];
        }
        $ensureHost($headers,$servername);
        $p['httpupgrade-opts']=[
            'path'=>$transport['path']??'/',
            'headers'=>$headers,
        ];
    }

    elseif($type==='xhttp'){
        $p['network']='xhttp';
        $headers=$normalizeHeaders($transport['headers']??[]);
        if(!empty($transport['host'])){
            $headers['Host']=is_array($transport['host'])?reset($transport['host']):$transport['host'];
        }
        $ensureHost($headers,$servername);
        $p['xhttp-opts']=[
            'mode'=>$transport['mode']??'auto',
            'path'=>$transport['path']??'/',
            'headers'=>$headers,
        ];
        if(!empty($transport['extra'])) $p['xhttp-opts']['extra']=$transport['extra'];
    }

    elseif($type==='quic'){
        $p['network']='quic';
    }

    elseif($type==='tcp'){
        $p['network']='tcp';
        if(($transport['header']['type']??'')==='http'){
            $req=$transport['header']['request']??[];
            $headers=$normalizeHeaders($req['headers']??[]);
            $ensureHost($headers,$servername);
            $p['network']='http';
            $p['http-opts']=[
                'method'=>'GET',
                'path'=>$req['path']??['/'],
                'headers'=>$headers,
            ];
        }
    }

    // ---------- FINAL CLEAN ----------
    return array_filter($p,function($v){
        if($v===null) return false;
        if($v==='') return false;
        if(is_array($v)&&empty($v)) return false;
        return true;
    });
}

function looks_like_base64($s): bool {
  $t = trim((string)$s);
  if($t === '' || strlen($t) < 32) return false;
  $t = preg_replace('/\s+/', '', $t);
  if(!preg_match('~^[A-Za-z0-9+/=]+$~', $t)) return false;
  $d = base64_decode($t, true);
  return ($d !== false && $d !== '');
}

function maybe_decode_base64($body): string {
  if(!looks_like_base64($body)) return $body;
  $t = preg_replace('/\s+/', '', trim((string)$body));
  $d = base64_decode($t, true);
  return ($d !== false && $d !== '') ? $d : $body;
}

function body_looks_like_json($body): bool {
  $t = preg_replace('/^\xEF\xBB\xBF/', '', (string)$body);
  $t = ltrim($t);
  return $t !== '' && ($t[0] === '{' || $t[0] === '[');
}

function b2h($b) { 
    if($b===0) return '0 B';
    $u=['B','KB','MB','GB','TB']; $i=0;
    while($b>=1024 && $i<4){ $b/=1024; $i++; }
    return round($b,2).' '.$u[$i];
}

function parse_query_map($q): array {
  $out = [];
  if($q === null || $q === '') return $out;
  foreach(explode('&', $q) as $pair){
    if($pair === '') continue;
    $kv = explode('=', $pair, 2);
    $k = urldecode($kv[0]);
    $v = isset($kv[1]) ? urldecode($kv[1]) : '';
    if($k !== '') $out[$k] = $v;
  }
  return $out;
}
function vless_link_to_singbox_outbound(string $line, string $defaultTag): ?array {
  $p = @parse_url($line);
  if(!$p || empty($p['scheme']) || strtolower($p['scheme']) !== 'vless') return null;
  
  $uuid = isset($p['user']) ? (string)$p['user'] : '';
  $server = (string)($p['host'] ?? '');
  $port = (int)($p['port'] ?? 0);
  
  if($server === '' || $port <= 0 || $uuid === '') return null;
  
  // Fix: Correctly decode the fragment (tag)
  $tag = $defaultTag;
  if(isset($p['fragment']) && $p['fragment'] !== ''){
    $tag = urldecode($p['fragment']);
  }
  
  $q = parse_query_map($p['query'] ?? '');
  $ob = [
    'type' => 'vless',
    'tag' => $tag,
    'server' => $server,
    'server_port' => $port,
    'uuid' => $uuid,
  ];

  if(!empty($q['flow'])) $ob['flow'] = (string)$q['flow'];

  $t = strtolower((string)($q['type'] ?? $q['transport'] ?? 'tcp'));
  
  // --- Transport Parsing ---
  if($t === 'ws'){
    $tr = ['type'=>'ws'];
    if(!empty($q['path'])) $tr['path'] = (string)$q['path'];
    if(!empty($q['host'])) $tr['headers'] = ['Host'=>(string)$q['host']];
    $ob['transport'] = $tr;
  } 
  elseif($t === 'grpc'){
    $tr = ['type'=>'grpc'];
    $sn = (string)($q['serviceName'] ?? $q['service_name'] ?? '');
    if($sn !== '') $tr['service_name'] = $sn;
    if(!empty($q['mode'])) $tr['mode'] = (string)$q['mode'];
    $ob['transport'] = $tr;
  } 
  elseif($t === 'xhttp' || $t === 'httpupgrade'){
    $tr = ['type' => $t]; 
    if(!empty($q['path'])) $tr['path'] = (string)$q['path'];
    if(!empty($q['host'])) $tr['host'] = [(string)$q['host']];
    if(!empty($q['mode'])) $tr['mode'] = (string)$q['mode'];
    if(!empty($q['extra'])) $tr['extra'] = (string)$q['extra'];
    $ob['transport'] = $tr;
  } 
  else {
    $tr = ['type'=>'tcp'];
    $headerType = (string)($q['headerType'] ?? '');
    if($headerType !== '' && $headerType !== 'none') {
        $tr['header'] = ['type' => $headerType];
        $req = [];
        if(!empty($q['path'])) $req['path'] = [(string)$q['path']];
        if(!empty($q['host'])) $req['headers'] = ['Host' => [(string)$q['host']]];
        if(!empty($req)) $tr['header']['request'] = $req;
    }
    $ob['transport'] = $tr;
  }

  // --- Security / TLS Parsing ---
  $security = strtolower((string)($q['security'] ?? ''));
  $tlsEnabled = ($security === 'tls' || $security === 'reality');
  
  if($tlsEnabled){
    $tls = ['enabled'=>true];
    
    // SNI priority: sni > serverName > servername
    $sni = (string)($q['sni'] ?? $q['serverName'] ?? $q['servername'] ?? '');

    $fp = (string)($q['fp'] ?? '');
    if($fp !== '') $tls['utls'] = ['enabled'=>true, 'fingerprint'=>$fp];
    
    $alpn = (string)($q['alpn'] ?? '');
    if ($alpn !== '') $tls['alpn'] = explode(',', $alpn);
    
    if($security === 'reality'){
      $reality = ['enabled' => true];
      $pbk = (string)($q['pbk'] ?? $q['publicKey'] ?? '');
      $sid = (string)($q['sid'] ?? $q['shortId'] ?? '');
      if($pbk !== '') $reality['public_key'] = $pbk;
      if($sid !== '') $reality['short_id'] = $sid;
      $tls['reality'] = $reality;
    }
    $ob['tls'] = $tls;
  }
  return $ob;

function vmess_link_to_singbox_outbound(string $line, string $defaultTag): ?array {
  $raw = trim($line);
  if(stripos($raw, 'vmess://') !== 0) return null;
  $b64 = substr($raw, 8);
  $json = base64_decode($b64, true);
  if($json === false) return null;
  $cfg = json_decode($json, true);
  if(!is_array($cfg)) return null;
  
  $tag = (string)($cfg['ps'] ?? $defaultTag);
  $server = (string)($cfg['add'] ?? '');
  $port = (int)($cfg['port'] ?? 0);
  $uuid = (string)($cfg['id'] ?? '');
  if($server === '' || $port <= 0 || $uuid === '') return null;
  
  $ob = [
    'type' => 'vmess',
    'tag' => $tag,
    'server' => $server,
    'server_port' => $port,
    'uuid' => $uuid,
    'security' => 'auto',
  ];

  $net = strtolower((string)($cfg['net'] ?? 'tcp'));
  
  if($net === 'ws'){
    $tr = ['type'=>'ws'];
    if(!empty($cfg['path'])) $tr['path'] = (string)$cfg['path'];
    if(!empty($cfg['host'])) $tr['headers'] = ['Host'=>(string)$cfg['host']];
    $ob['transport'] = $tr;
  } elseif($net === 'grpc'){
    $tr = ['type'=>'grpc'];
    if(!empty($cfg['path'])) $tr['service_name'] = (string)$cfg['path'];
    $ob['transport'] = $tr;
  } elseif($net === 'xhttp' || $net === 'httpupgrade') {
    // --- XHTTP Support ---
    $tr = ['type'=>'xhttp'];
    if(!empty($cfg['path'])) $tr['path'] = (string)$cfg['path'];
    if(!empty($cfg['host'])) $tr['host'] = [(string)$cfg['host']];
    // ADDED: Support for XHTTP Mode and Extra
    if(!empty($cfg['mode'])) $tr['mode'] = (string)$cfg['mode'];
    if(!empty($cfg['extra'])) $tr['extra'] = (string)$cfg['extra'];
    $ob['transport'] = $tr;
  } elseif($net === 'tcp') {

    $tr = ['type' => 'tcp'];
    $headerType = strtolower((string)($cfg['type'] ?? ''));
    if ($headerType === 'http') {
        $tr['header'] = [
            'type' => 'http',
            'request' => [
                'path' => [(!empty($cfg['path']) ? (string)$cfg['path'] : '/')],
                'headers' => [
                    'Host' => [(!empty($cfg['host']) ? (string)$cfg['host'] : $server)]
                ]
            ]
        ];
    }
    $ob['transport'] = $tr;
  }

  $tlsFlag = strtolower((string)($cfg['tls'] ?? ''));
  if($tlsFlag === 'tls'){
    $tls = ['enabled'=>true];
    if(!empty($cfg['sni'])) $tls['server_name'] = (string)$cfg['sni'];
    $ob['tls'] = $tls;
  }
  return $ob;
}


// --- CONVERTER: XRAY/V2RAY JSON -> SINGBOX INTERNAL ---
function xray_json_to_singbox($vObj, $tag) {
    $protocol = strtolower($vObj['protocol'] ?? '');
    $out = ['tag' => $tag];
    
    $settings = $vObj['settings'] ?? [];
    $stream = $vObj['streamSettings'] ?? [];
    
    if (isset($settings['vnext']) && is_array($settings['vnext']) && isset($settings['vnext'][0])) {
        $vn = $settings['vnext'][0];
        $out['server'] = $vn['address'] ?? '';
        $out['server_port'] = $vn['port'] ?? 0;
        if (isset($vn['users']) && isset($vn['users'][0])) {
            $u = $vn['users'][0];
            $out['uuid'] = $u['id'] ?? $u['password'] ?? ''; 
            if(isset($u['flow'])) $out['flow'] = $u['flow'];
            if(isset($u['encryption'])) $out['security'] = $u['encryption'];
        }
    } elseif ($protocol === 'shadowsocks' && isset($settings['servers'][0])) {
        $srv = $settings['servers'][0];
        $out['server'] = $srv['address'] ?? '';
        $out['server_port'] = $srv['port'] ?? 0;
        $out['password'] = $srv['password'] ?? '';
        $out['method'] = $srv['method'] ?? '';
    } elseif (isset($settings['servers']) && isset($settings['servers'][0])) {
        $srv = $settings['servers'][0];
        $out['server'] = $srv['address'] ?? '';
        $out['server_port'] = $srv['port'] ?? 0;
        if(isset($srv['password'])) $out['password'] = $srv['password'];
    } else {
        return null;
    }

    if(empty($out['server'])) return null;

    if ($protocol === 'vless') $out['type'] = 'vless';
    elseif ($protocol === 'vmess') $out['type'] = 'vmess';
    elseif ($protocol === 'trojan') { 
        $out['type'] = 'trojan'; 
        if(empty($out['password']) && !empty($out['uuid'])) $out['password'] = $out['uuid'];
        unset($out['uuid']); 
    }
    elseif ($protocol === 'shadowsocks') $out['type'] = 'shadowsocks';
    else return null;

    $net = $stream['network'] ?? 'tcp';
    $out['transport'] = ['type' => $net];
    
    if ($net === 'ws') {
        $ws = $stream['wsSettings'] ?? [];
        if(!empty($ws['path'])) $out['transport']['path'] = $ws['path'];
        if(!empty($ws['headers']['Host'])) $out['transport']['headers'] = ['Host' => $ws['headers']['Host']];
    }
      // ADDED: Proper JSON extraction for XHTTP Settings
    if ($net === 'xhttp' || $net === 'httpupgrade') {
        $out['transport']['type'] = $net; // Preserve the exact protocol

        $xhttp = $stream['xhttpSettings'] ?? $stream['httpupgradeSettings'] ?? [];
        if(!empty($xhttp['path'])) $out['transport']['path'] = $xhttp['path'];
        if(!empty($xhttp['host'])) $out['transport']['host'] = [$xhttp['host']];
        if(!empty($xhttp['mode'])) $out['transport']['mode'] = $xhttp['mode'];
        if(!empty($xhttp['extra'])) $out['transport']['extra'] = is_array($xhttp['extra']) ? json_encode($xhttp['extra']) : $xhttp['extra'];
    }
    if ($net === 'grpc') {

        $grpc = $stream['grpcSettings'] ?? [];
        if(!empty($grpc['serviceName'])) $out['transport']['service_name'] = $grpc['serviceName'];
    }
if ($net === 'tcp') {
    $tcp = $stream['tcpSettings']['header'] ?? [];
    if (!empty($tcp) && is_array($tcp)) {
        $out['transport']['header'] = $tcp; // store raw header info
    }
}
    $security = $stream['security'] ?? 'none';
    if ($security === 'tls') {
        $tls = $stream['tlsSettings'] ?? [];
        $out['tls'] = ['enabled' => true];
        if(!empty($tls['serverName'])) $out['tls']['server_name'] = $tls['serverName'];
        if(!empty($tls['fingerprint'])) $out['tls']['utls'] = ['enabled'=>true, 'fingerprint'=>$tls['fingerprint']];
        if(!empty($tls['alpn'])) $out['tls']['alpn'] = $tls['alpn'];
    } elseif ($security === 'reality') {
        $reality = $stream['realitySettings'] ?? [];
        $out['tls'] = [
            'enabled' => true,
            'server_name' => $reality['serverName'] ?? '',
            'reality' => [
                'enabled' => true,
                'public_key' => $reality['publicKey'] ?? '',
                'short_id' => $reality['shortId'] ?? '',
            ],
            'utls' => ['enabled'=>true, 'fingerprint'=>$reality['fingerprint'] ?? 'chrome']
        ];
    }

    return $out;
}

// ---------------- RECONSTRUCTOR (FIXED: ALPN SUPPORT) ----------------
function reconstruct_link_from_json($ob) {
    $type = $ob['type'] ?? '';
    // Fix: Decode the tag first to avoid double encoding, then encode at the end
    $tagName = urldecode($ob['tag'] ?? 'Proxy');
    $tag = rawurlencode($tagName);
    
    // --- VLESS RECONSTRUCTION ---
    if ($type === 'vless') {
        $user = $ob['uuid'] ?? '';
        $server = $ob['server'] ?? '';
        $port = $ob['server_port'] ?? 443;
        
        // Standard VLESS parameters
        $params = [
            'encryption' => 'none', // MANDATORY for standard VLESS links
        ];
        
        if (($ob['tls']['enabled'] ?? false)) {
            $params['security'] = 'tls';
       $sni = detect_real_sni($ob);
if ($sni !== '') {
    $params['sni'] = $sni;
}
            $params['fp'] = $ob['tls']['utls']['fingerprint'] ?? 'chrome';
            
            // Fix: Handle ALPN correctly
            if (!empty($ob['tls']['alpn'])) {
                $params['alpn'] = is_array($ob['tls']['alpn']) ? implode(',', $ob['tls']['alpn']) : $ob['tls']['alpn'];
            }

            if (($ob['tls']['reality']['enabled'] ?? false)) {
                $params['security'] = 'reality';
                $params['pbk'] = $ob['tls']['reality']['public_key'] ?? '';
                $params['sid'] = $ob['tls']['reality']['short_id'] ?? '';
            }
        } else {
             $params['security'] = 'none';
        }
        
        $transport = $ob['transport']['type'] ?? 'tcp';
        $params['type'] = $transport;

        // Transport Specific Logic
        if ($transport === 'http' || $transport === 'xhttp' || $transport === 'httpupgrade') {
             $params['type'] = ($transport === 'http') ? 'tcp' : $transport; // V2RayNG uses tcp+headerType=http
             $params['path'] = $ob['transport']['path'] ?? '/';

             $h = $ob['transport']['host'] ?? [];
             if(!empty($h)) $params['host'] = is_array($h) ? $h[0] : $h;
             
             if(!empty($ob['transport']['mode'])) $params['mode'] = $ob['transport']['mode'];
             if(!empty($ob['transport']['extra'])) $params['extra'] = $ob['transport']['extra'];
        }
        elseif ($transport === 'ws') {
            $params['path'] = $ob['transport']['path'] ?? '/';
            // CRITICAL FIX: Do NOT fallback to SNI. Only set host if headers exist.
            if(!empty($ob['transport']['headers']['Host'])) {
                $params['host'] = $ob['transport']['headers']['Host'];
            }
        } 
        elseif ($transport === 'grpc') {
            $params['serviceName'] = $ob['transport']['service_name'] ?? '';
            $params['mode'] = $ob['transport']['mode'] ?? 'gun';
        }
        elseif ($transport === 'tcp') {
            if(!empty($ob['transport']['header'])) {
                $params['headerType'] = $ob['transport']['header']['type'] ?? 'none';
                $req = $ob['transport']['header']['request'] ?? [];
                
                // Fix Path
                if (!empty($req['path'])) {
                    $params['path'] = is_array($req['path']) ? $req['path'][0] : $req['path'];
                }
                
                // Fix Host (Check uppercase Host and lowercase host)
                $hostVal = $req['headers']['Host'] ?? $req['headers']['host'] ?? null;
                if (!empty($hostVal)) {
                    $params['host'] = is_array($hostVal) ? $hostVal[0] : $hostVal;
                }
            }
        }

        
        if(!empty($ob['flow'])) $params['flow'] = $ob['flow'];
        
        // Remove empty parameters to keep link clean
        $query = http_build_query(array_filter($params, function($v){ return $v !== ''; }));
        return "vless://$user@$server:$port?$query#$tag";
    }
    
    // --- VMESS RECONSTRUCTION ---
    if ($type === 'vmess') {
        $transport = $ob['transport']['type'] ?? 'tcp';
        $v = [
            "v"    => "2", 
            "ps"   => $tagName, // VMess JSON uses raw string, not encoded
            "add"  => $ob['server'],
            "port" => (int)$ob['server_port'], 
            "id"   => $ob['uuid'], 
            "aid"  => (int)($ob['alter_id'] ?? 0),
            "net"  => $transport, 
            "type" => "none", 
            "host" => "", 
            "path" => "",
            "tls"  => ($ob['tls']['enabled'] ?? false) ? "tls" : ""
        ];
        
       $sni = detect_real_sni($ob);
if ($sni !== '') {
    $v['sni'] = $sni;
}
        if (!empty($ob['tls']['alpn'])) {
             $v['alpn'] = is_array($ob['tls']['alpn']) ? implode(',', $ob['tls']['alpn']) : $ob['tls']['alpn'];
        }
        if (!empty($ob['tls']['utls']['fingerprint'])) $v['fp'] = $ob['tls']['utls']['fingerprint'];

        if ($transport === 'ws') {
            $v['path'] = $ob['transport']['path'] ?? '/';
            if(!empty($ob['transport']['headers']['Host'])) {
                 $v['host'] = $ob['transport']['headers']['Host'];
            }
        } 
        elseif ($transport === 'grpc') {
            $v['path'] = $ob['transport']['service_name'] ?? "";
        }
        elseif ($transport === 'xhttp' || $transport === 'httpupgrade') {
            $v['path'] = $ob['transport']['path'] ?? '/';
            if(!empty($ob['transport']['host'])) $v['host'] = is_array($ob['transport']['host']) ? $ob['transport']['host'][0] : $ob['transport']['host'];
            if(!empty($ob['transport']['mode'])) $v['mode'] = $ob['transport']['mode'];
            if(!empty($ob['transport']['extra'])) $v['extra'] = $ob['transport']['extra'];
        }
        elseif ($transport === 'tcp' && !empty($ob['transport']['header'])) {
             $h = $ob['transport']['header'];
             $v['type'] = $h['type'] ?? 'none';
             if ($v['type'] === 'http' && !empty($h['request'])) {
                 $v['path'] = $h['request']['path'][0] ?? "";
                 $v['host'] = $h['request']['headers']['Host'][0] ?? "";
             }
        } 
        return "vmess://" . base64_encode(json_encode($v));
    }
    return null; 
}

function clash_safe_name($s) {
    // Remove non-ASCII except space and basic symbols
    $s = preg_replace('/[^\x20-\x7E]/', '', $s);
    $s = trim($s);
    if ($s === '') $s = 'Node';
    return $s;
}
function fetch_upstream_raw_multi(array $urls, bool $singboxUA=false): array {
    $ua = $singboxUA ? 'sing-box/1.11.0' : 'v2rayNG/1.8.5'; 
    $accept = $singboxUA ? 'Accept: application/json, text/plain;q=0.9, */*;q=0.8' : 'Accept: */*';

    // LOG URLS
    foreach ($urls as $url) {
        debug_log("Fetching upstream URL", $url);
    }

    $mh = curl_multi_init();
    $chs = [];

    foreach ($urls as $url) {
        $ch = curl_init($url);
        curl_setopt_array($ch, [
            CURLOPT_RETURNTRANSFER => true,
            CURLOPT_HEADER         => true,
            CURLOPT_FOLLOWLOCATION => true,
            CURLOPT_MAXREDIRS      => 3,
            CURLOPT_CONNECTTIMEOUT => 5, 
            CURLOPT_TIMEOUT        => 15,
            CURLOPT_ENCODING       => '',
            CURLOPT_USERAGENT      => $ua,
            CURLOPT_HTTPHEADER     => [$accept,'Cache-Control: no-cache','Pragma: no-cache'],
            CURLOPT_SSL_VERIFYHOST => 0,
            CURLOPT_SSL_VERIFYPEER => 0,
        ]);
        curl_multi_add_handle($mh, $ch);
        $chs[] = $ch;
    }

    $running = null;
    do {
        curl_multi_exec($mh, $running);
        curl_multi_select($mh, 0.1);
    } while ($running > 0);

    $out = [];
    foreach ($chs as $ch) {
        $resp = curl_multi_getcontent($ch);
        $code = (int)curl_getinfo($ch, CURLINFO_RESPONSE_CODE);
        $hs   = (int)curl_getinfo($ch, CURLINFO_HEADER_SIZE);

        if ($hs > 0) {
            $hdrs = substr((string)$resp, 0, $hs);
            $body = substr((string)$resp, $hs);
        } else {
            $body = (string)$resp;
        }

        // LOG RESPONSE STATUS
        debug_log("Upstream response code", ['code'=>$code,'body_length'=>strlen($body)]);

        $out[] = [
            'ok' => ($code>=200 && $code<300),
            'body' => $body,
        ];

        curl_multi_remove_handle($mh, $ch);
        curl_close($ch);
    }

    curl_multi_close($mh);
    return $out;
}
