if(!function_exists('persianDate')) {
    function persianDate($format, $timestamp) {
        if(!$timestamp) return '∞';
        $g_d = date('j', $timestamp); $g_m = date('n', $timestamp); $g_y = date('Y', $timestamp);
        $d_4 = $g_y % 4; $g_a = array(0,0,31,59,90,120,151,181,212,243,273,304,334);
        $doy_g = $g_a[$g_m] + $g_d; if($d_4 == 0 && $g_m > 2) $doy_g++;
        $d_33 = (int)((($g_y - 16) % 132) * 0.0305); $a = ($d_33 == 3 || $d_33 < ($d_4 - 1) || $d_4 == 0) ? 286 : 287;
        if($doy_g > 79) { $j_y = $g_y - 621; $doy_j = $doy_g - 79; } 
        else { $j_y = $g_y - 622; $doy_j = $doy_g + $a; }
        if($doy_j < 187) { $j_m = (int)(($doy_j - 1) / 31); $j_d = $doy_j - (31 * $j_m); $j_m++; } 
        else { $j_m = (int)(($doy_j - 187) / 30); $j_d = $doy_j - 186 - ($j_m * 30); $j_m += 7; }
        return $j_y . '/' . sprintf('%02d', $j_m) . '/' . sprintf('%02d', $j_d);
    }
}
if(!function_exists('formatBytes')) {
    function formatBytes($bytes) { 
        if($bytes == 0) return '0 MB';
        $i = floor(log($bytes, 1024));
        return round($bytes / pow(1024, $i), 2) . ' ' . ['B','KB','MB','GB','TB'][$i];
    }
}

// ---> NEW FEATURE: RESET SUBSCRIPTION TOKEN <---
if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['reset_token'])) {
    try {
        $pdo->beginTransaction();
        $stmtCheck = $pdo->prepare("SELECT rst.token FROM reseller_services rs LEFT JOIN reseller_sub_tokens rst ON rs.id = rst.service_id WHERE rs.id = ? AND rs.reseller_tg_id = ?");
        $stmtCheck->execute([$id, $rid]);
        $svcData = $stmtCheck->fetch();

        if ($svcData && !empty($svcData['token'])) {
            $oldToken = $svcData['token'];
            // Generate a new 32-character secure token
            $newToken = bin2hex(random_bytes(16));

            // Update token in sub_tokens table
            $stmtUpdateToken = $pdo->prepare("UPDATE reseller_sub_tokens SET token = ? WHERE service_id = ?");
            $stmtUpdateToken->execute([$newToken, $id]);

            // Update token in cache to prevent user from losing their usage stats temporarily
            $stmtUpdateCache = $pdo->prepare("UPDATE user_stats_cache SET token = ? WHERE token = ?");
            $stmtUpdateCache->execute([$newToken, $oldToken]);

            $pdo->commit();
            $msg_alert = "<div class='mac-alert success'><span class='icon'>check_circle</span><span>لینک اتصال تغییر یافت! لینک قبلی کاربر باطل شد.</span></div>";
            echo "<script>setTimeout(()=>window.location.href='?page=detail&id=$id', 1500);</script>";
        } else { 
            throw new Exception("توکن قبلی یافت نشد."); 
        }
    } catch (Exception $e) { 
        if($pdo->inTransaction()) $pdo->rollBack(); 
        $msg_alert = "<div class='mac-alert error'><span class='icon'>error</span><span>خطا: " . $e->getMessage() . "</span></div>";

                // If the service is active and the user requested "Next Plan"
                if ($targetSvc['status'] === 'active' && $autoRenew) {
                    $stmtUpdate = $pdo->prepare("UPDATE reseller_services SET auto_renew = 1, next_plan_gb = ?, next_plan_days = ? WHERE id = ? AND reseller_tg_id = ?");
                    $stmtUpdate->execute([$newGb, $newDays, $id, $rid]);
                    
                    // Deduct from reseller balance
                    $stmtDeduct = $pdo->prepare("UPDATE resellers SET used_gb = used_gb + ? WHERE tg_id=?");
                    $stmtDeduct->execute([$newGb, $rid]);
                    
                    // Add to sales log
                    $stmtLog = $pdo->prepare("INSERT INTO reseller_sales_log (reseller_tg_id, service_id, action_type, gb_amount, days_amount) VALUES (?, ?, 'renew', ?, ?)");
                    $stmtLog->execute([$rid, $id, $newGb, $newDays]);
                    
                    $pdo->commit();

                    echo "<script>setTimeout(()=>window.location.href='?page=detail&id=$id', 1500);</script>";
                } 
                // Immediate Renewal
                else {
                    $newExpireTs = time() + ($newDays * 86400);
                    $newExpireDate = date('Y-m-d H:i:s', $newExpireTs);
                    $newTotalBytes = $newGb * 1024 * 1024 * 1024;
                    
                    // Clear next plan if we are doing an immediate manual renewal
                    $stmtUpdate = $pdo->prepare("UPDATE reseller_services SET gb=?, days=?, expire_at=?, status='active', last_used_bytes=0, auto_renew=0, next_plan_gb=NULL, next_plan_days=NULL WHERE id=? AND reseller_tg_id=?");
                    $stmtUpdate->execute([$newGb, $newDays, $newExpireDate, $id, $rid]);
                    
                    $stmtDeduct = $pdo->prepare("UPDATE resellers SET used_gb = used_gb + ? WHERE tg_id=?");
                    $stmtDeduct->execute([$newGb, $rid]);
                    
                    // Sync Queue
                    $payload = json_encode(['gb' => $newGb, 'days' => $newDays, 'expire_ts' => $newExpireTs, 'enable' => true]);
                    $stmtQueue = $pdo->prepare("INSERT INTO reseller_sync_queue (service_id, username, action, payload, status) VALUES (?, ?, 'renew', ?, 'pending')");
                    $stmtQueue->execute([$id, $targetSvc['rain_username'], $payload]);
                    
                    $stmtLog = $pdo->prepare("INSERT INTO reseller_sales_log (reseller_tg_id, service_id, action_type, gb_amount, days_amount) VALUES (?, ?, 'renew', ?, ?)");
                    $stmtLog->execute([$rid, $id, $newGb, $newDays]);
                    
                    // Cache Reset
                    $stmtToken = $pdo->prepare("SELECT token FROM reseller_sub_tokens WHERE service_id = ? LIMIT 1");
                    $stmtToken->execute([$id]);
                    $tokenRow = $stmtToken->fetch();
                    if ($tokenRow) {
                        $stmtCache = $pdo->prepare("UPDATE user_stats_cache SET used_bytes = 0, total_bytes = ?, expire_ts = ?, is_disabled = 0, last_updated = UNIX_TIMESTAMP() WHERE token = ?");
                        $stmtCache->execute([$newTotalBytes, $newExpireTs, $tokenRow['token']]);
                    }
                    $pdo->commit();
                    $msg_alert = "<div class='mac-alert success'><span class='icon'>check_circle</span><span>سرویس با موفقیت تمدید و به شبکه ارسال شد.</span></div>";
                    echo "<script>setTimeout(()=>window.location.href='?page=detail&id=$id', 1500);</script>";
                }

    function calculateSuggestion() {
        const days = parseInt(renewDaysInput.value) || 0;
        
        if (days > 0) {
            suggestionBox.classList.remove('hidden');
            
            // Profile Detect
            let profileName = ""; let profileColor = "";
            if (userDailyAvg < 0.5) { profileName = "کم‌مصرف"; profileColor = "#34C759"; } 
            else if (userDailyAvg < 1.5) { profileName = "معمولی"; profileColor = "#007AFF"; } 
            else if (userDailyAvg < 3.0) { profileName = "پرمصرف"; profileColor = "#FF9500"; } 
            else { profileName = "فوق پرمصرف"; profileColor = "#FF2D55"; }
            
            userProfileText.innerHTML = `<span style="color:${profileColor}; font-weight:800; font-size:13px;">${profileName}</span> <span style="color:#8E8E93; font-size:11px; font-family:monospace;">(${userDailyAvg} GB/day)</span>`;

            // Calculate Suggestion
            let pureUsage = days * userDailyAvg;
            let predictedUsage = pureUsage * 1.20; 
            let suggestedGb = Math.ceil(predictedUsage / 5) * 5; 
            if (suggestedGb < 10) suggestedGb = 10; 
            suggestGbText.innerText = suggestedGb + " GB";
            
            // Financials
            let resDiscount = 0;
            if (suggestedGb >= 80) { resDiscount = Math.floor((suggestedGb - 60) / 20) * 15; }
            
            let rawBuyPrice = suggestedGb * adminBaseSell;
            let finalRawPrice = rawBuyPrice * (1 - (resDiscount / 100));
            
            let minAdminSellPrice = suggestedGb * adminMinSell;
            if (finalRawPrice < minAdminSellPrice) { 
                finalRawPrice = minAdminSellPrice; 
                resDiscount = parseFloat((100 - ((finalRawPrice / rawBuyPrice) * 100)).toFixed(1)); 
            }

            let custDiscount = resDiscount; 
            let baseSellPrice = suggestedGb * perGigSellingPrice;
            let finalSellPrice = baseSellPrice * (1 - (custDiscount / 100));
            let finalProfit = finalSellPrice - finalRawPrice;
            
            // Render
            let rawBadge = resDiscount > 0 ? ` <span class="text-[10px] bg-[#FF2D55]/10 text-[#FF2D55] px-1.5 py-0.5 rounded ml-1 font-bold">-${resDiscount}٪</span>` : '';

                if(diff < 60) timeStr = "JUST NOW";
                else if(diff < 3600) timeStr = Math.floor(diff/60) + " MINS AGO";
                else if(diff < 86400) timeStr = Math.floor(diff/3600) + " HOURS AGO";
                else timeStr = Math.floor(diff/86400) + " DAYS AGO";
            }
            document.getElementById('conn-time').innerText = timeStr;
        } else {
            const row = document.createElement('div');
            row.className = 'log-line text-[#8E8E93] opacity-70';
            row.innerHTML = `> REPORT: USER OFFLINE`;
            logsEl.prepend(row);
        }
    }

    if(allKeys.length > 0) setTimeout(startScan, 500); // Start scan slightly after page load
</script>
