function utcToTehranTimestamp($utcString) {
    if (empty($utcString)) return time();
    $dt = new DateTime($utcString, new DateTimeZone('UTC'));
    $dt->setTimezone(new DateTimeZone('Asia/Tehran'));
    return $dt->getTimestamp();
}

function stringToColor($str) {
    $code = dechex(crc32($str));
    $code = substr($code, 0, 6);
    return '#' . $code;
}

function getInitials($name) {
    $name = trim($name);
    if (empty($name)) return '?';
    $parts = explode(' ', $name);
    if (count($parts) > 1) {
        return mb_substr($parts[0], 0, 1) . mb_substr($parts[1], 0, 1);
    }
    return mb_substr($name, 0, 2);
}

// SMART PRICING FORMULA
function calculateSmartPricing($gb, $resellerBasePrice, $adminBase, $adminMin) {
    $reseller_discount_pct = 0;
    
    if ($gb >= 80) { 
        $reseller_discount_tier = floor(($gb - 60) / 20);
        $reseller_discount_pct = $reseller_discount_tier * 15;
    }
    
    $raw_buy_price = $gb * $adminBase;
    $reseller_buy_price = $raw_buy_price * (1 - ($reseller_discount_pct / 100));
    
    $min_admin_sell_price = $gb * $adminMin;
    
    if ($reseller_buy_price < $min_admin_sell_price) {
        $reseller_buy_price = $min_admin_sell_price;
        $reseller_discount_pct = round(100 - (($reseller_buy_price / $raw_buy_price) * 100), 1);
    }

    $customer_discount_pct = $reseller_discount_pct;
    
    $raw_sell_price = $gb * $resellerBasePrice;
    $customer_sell_price = $raw_sell_price * (1 - ($customer_discount_pct / 100));
    
    $net_profit = $customer_sell_price - $reseller_buy_price;

    return [
        'buy_price' => $reseller_buy_price,
        'res_discount' => $reseller_discount_pct,
        'sell_price' => $customer_sell_price,
        'cust_discount' => $customer_discount_pct,
        'profit' => $net_profit
    ];
}

<script>
// --- CURRENCY FORMATTER FUNCTION ---
function formatNumberWithCommas(value) {
    if (!value) return '';
    const numericValue = value.replace(/\D/g, ''); // strip all non-digits
    return numericValue.replace(/\B(?=(\d{3})+(?!\d))/g, ",");
}

// Run formatter on all inputs with class 'price-format' on page load and on keyup
document.addEventListener('DOMContentLoaded', () => {
    document.querySelectorAll('.price-format').forEach(input => {
        // Init load
        input.value = formatNumberWithCommas(input.value);

        // While typing
        input.addEventListener('input', function(e) {
            let cursorPosition = this.selectionStart;
            let originalLength = this.value.length;
            
            this.value = formatNumberWithCommas(this.value);
            
            let newLength = this.value.length;
            cursorPosition = cursorPosition + (newLength - originalLength);
            this.setSelectionRange(cursorPosition, cursorPosition);
        });
    });
});

// --- AJAX FUNCTION TO SAVE OVERRIDE PRICE & PAID TOGGLE ---
function updateSaleData(saleId, context) {
    let priceInputRaw = document.getElementById('price_' + context + '_' + saleId).value;
    let priceInput = priceInputRaw.replace(/,/g, ''); // Strip commas before sending to DB
    
    let isPaid = document.getElementById('paid_' + context + '_' + saleId).checked ? 1 : 0;
    
    let formData = new FormData();
    formData.append('ajax_update_sale', '1');
    formData.append('sale_id', saleId);
    formData.append('custom_price', priceInput);
    formData.append('is_paid', isPaid);
    
    fetch(window.location.href, {
        method: 'POST',
        body: formData
    })
    .then(response => response.json())
    .then(data => {
        if(data.success) {
            console.log('Saved sale #' + saleId);
            // Refresh to update the profit calculation in UI automatically
            setTimeout(() => { window.location.reload(); }, 200);
        }
    })
    .catch(error => console.error('Error:', error));
}

// --- PRINT FUNCTION ---
function triggerSmartPrint() {
    const printContent = document.getElementById('print-template-container').innerHTML;
    const printStyles = document.getElementById('isolated-print-styles').innerHTML;
    
    const iframe = document.createElement('iframe');
    iframe.style.position = 'fixed';
    iframe.style.right = '0';
    iframe.style.bottom = '0';
    iframe.style.width = '0';
    iframe.style.height = '0';
    iframe.style.border = 'none';
    document.body.appendChild(iframe);
    
    const doc = iframe.contentWindow.document;
    doc.open();
    doc.write(`
        <!DOCTYPE html>
        <html lang="fa" dir="rtl">
        <head>
            <meta charset="UTF-8">
            <title>فاکتور فروش و حسابداری</title>
            <style>${printStyles}</style>
        </head>
        <body>
            ${printContent}
        </body>
        </html>
    `);
    doc.close();
    
    setTimeout(() => {
        iframe.contentWindow.focus();
        iframe.contentWindow.print();
        setTimeout(() => { document.body.removeChild(iframe); }, 1000);
    }, 500); 
}
