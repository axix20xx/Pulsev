<?php
error_reporting(E_ALL);
ini_set('display_errors', '1');
$result = null;
function apiRequest($base, $path, $token = null) {
    $ch = curl_init(rtrim($base, '/') . $path);
    $headers = ['Accept: application/json'];
    if ($token) {
        $headers[] = 'Authorization: Bearer ' . $token;
    }
    curl_setopt_array($ch, [
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_HTTPHEADER => $headers,
        CURLOPT_CONNECTTIMEOUT => 10,
        CURLOPT_TIMEOUT => 30,
        CURLOPT_SSL_VERIFYPEER => true,
        CURLOPT_SSL_VERIFYHOST => 2,
    ]);
    $body = curl_exec($ch);
    $error = curl_error($ch);
    $status = curl_getinfo($ch, CURLINFO_HTTP_CODE);
    curl_close($ch);
    $json = json_decode($body ?: '', true);
    return [
        'http_status' => $status,
        'data' => $json ?? $body,
        'error' => $error ?: null,
    ];
}
function showResult($base, $path, $token) {
    return apiRequest($base, $path, $token);
}
if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $base = rtrim(trim($_POST['base'] ?? ''), '/');
    $username = trim($_POST['username'] ?? '');
    $password = $_POST['password'] ?? '';
    $customer = trim($_POST['customer'] ?? '');
    if (
        !filter_var($base, FILTER_VALIDATE_URL) ||
        !in_array(parse_url($base, PHP_URL_SCHEME), ['https', 'http'], true) ||
        !$username || !$password
    ) {
        $result = ['error' => 'Enter a valid panel URL, username and password.'];
    } else {
        // Authenticate using form-urlencoded credentials.
        $ch = curl_init($base . '/api/admin/token');
        curl_setopt_array($ch, [
            CURLOPT_POST => true,
            CURLOPT_POSTFIELDS => http_build_query([
                'username' => $username,
                'password' => $password,
            ]),
            CURLOPT_RETURNTRANSFER => true,
            CURLOPT_HTTPHEADER => [
                'Content-Type: application/x-www-form-urlencoded',
                'Accept: application/json',
            ],
            CURLOPT_CONNECTTIMEOUT => 10,
            CURLOPT_TIMEOUT => 30,
            CURLOPT_SSL_VERIFYPEER => true,
            CURLOPT_SSL_VERIFYHOST => 2,
        ]);
        $body = curl_exec($ch);
        $status = curl_getinfo($ch, CURLINFO_HTTP_CODE);
        $error = curl_error($ch);
        curl_close($ch);
        $auth = json_decode($body ?: '', true);
        if ($error || $status !== 200 || empty($auth['access_token'])) {
            $result = [
                'authentication' => [
                    'http_status' => $status,
                    'error' => $error ?: null,
                    'response' => $auth ?? $body,
                ],
            ];
        } else {
            $token = $auth['access_token'];
            $paths = [
                'Admin' => '/api/admin',
                'Core' => '/api/core',
                'System' => '/api/system',
                'Inbounds' => '/api/inbounds',
                'Users (first 100)' => '/api/users?offset=0&limit=100',
                'IPsec' => '/api/ipsec',
            ];
            $result = ['authentication' => 'Success'];
            foreach ($paths as $label => $path) {
                $result[$label] = showResult($base, $path, $token);
            }
            // Fetch configs for a specific customer if provided.
            if ($customer !== '') {
                $u = rawurlencode($customer);
                $user = showResult(
                    $base, "/api/user/$u", $token
                );
                $result['Customer'] = $user;
                $core = $user['data']['core'] ?? '';
                if ($user['http_status'] === 200) {
                    if ($core === 'xray') {
                        $configPaths = [
                            'Xray Link' =>
                                "/api/xray/user/$u/link",
                            'Xray Configs' =>
                                "/api/xray/user/$u/configs",
                        ];
                    } elseif ($core === 'wireguard') {
                        $configPaths = [
                            'WireGuard Configs' =>
                                "/api/wg/user/$u/configs",
                            'WireGuard Bundle' =>
                                "/api/wg/user/$u/bundle",
                        ];
                    } elseif ($core === 'openvpn') {
                        $configPaths = [
                            'OpenVPN Configs' =>
                                "/api/ovpn/user/$u/configs",
                            'OpenVPN Import' =>
                                "/api/ovpn/user/$u/import",
                        ];
                    } else {
                        $configPaths = [];
                        $result['Customer Config Error'] =
                            'Unknown or unsupported core.';
                    }
                    foreach ($configPaths as $label => $path) {
                        $result[$label] = showResult(
                            $base, $path, $token
                        );
                    }
                }
            }
        }
    }
}
