if(!function_exists('formatBytes')) {
    function formatBytes($bytes) { 
        if($bytes == 0) return '0 MB';
        $i = floor(log($bytes, 1024));
        return round($bytes / pow(1024, $i), 2) . ' ' . ['B','KB','MB','GB','TB'][$i];
    }
}
if(!function_exists('tsToJalali')) {
    function tsToJalali($timestamp) {
        if(!$timestamp || $timestamp <= 0) return 'نامحدود';
        
        $g_d=date('j',$timestamp); $g_m=date('n',$timestamp); $g_y=date('Y',$timestamp);
        $d_4=$g_y%4; $g_a=array(0,0,31,59,90,120,151,181,212,243,273,304,334);
        $doy_g=$g_a[$g_m]+$g_d; if($d_4==0 && $g_m>2)$doy_g++;
        $d_33=(int)((($g_y-16)%132)*.0305); $a=($d_33==3 || $d_33<($d_4-1) || $d_4==0)?286:287;
        $b=(($d_33==1 || $d_33==2) && ($d_33==$d_4 || $d_4==1))?19:20;
        if($doy_g>79){$j_y=$g_y-621; $doy_j=$doy_g-79;}else{$j_y=$g_y-622; $doy_j=$doy_g+$a;}
        if($doy_j<187){$j_m=(int)(($doy_j-1)/31); $j_d=$doy_j-(31*$j_m); $j_m++;}else{$j_m=(int)(($doy_j-187)/30); $j_d=$doy_j-186-($j_m*30); $j_m+=7;}
        return $j_y.'/'.sprintf('%02d',$j_m).'/'.sprintf('%02d',$j_d);
    }
}
if(!function_exists('relativeTime')) {
    function relativeTime($ts) {
        if(!$ts || $ts <= 0) return 'نامحدود';
        $diff = $ts - time();
        $absDiff = abs($diff);
        $units = [31536000 => 'سال', 2592000 => 'ماه', 604800 => 'هفته', 86400 => 'روز', 3600 => 'ساعت', 60 => 'دقیقه'];
        foreach ($units as $val => $str) {
            if ($absDiff >= $val) {
                $count = floor($absDiff / $val);
                $suffix = ($diff > 0) ? 'مانده' : 'پیش';
                return "$count $str $suffix";
            }
        }
        return ($diff > 0) ? 'همین الان' : 'منقضی شده';
    }
}

            CASE 
                WHEN usc.expire_ts IS NOT NULL AND usc.expire_ts > 0 THEN usc.expire_ts
                ELSE UNIX_TIMESTAMP(rs.expire_at)
            END AS final_expire_ts,
            DATEDIFF(NOW(), rs.created_at) as days_active
        FROM reseller_services rs
        LEFT JOIN reseller_customers rc ON rs.customer_id = rc.id
        LEFT JOIN reseller_sub_tokens rst ON rs.id = rst.service_id
        LEFT JOIN user_stats_cache usc ON rst.token COLLATE utf8mb4_unicode_ci = usc.token
        $whereSql
        $orderSql

<script>
    // --- 1. SOLVE THE CSS TRANSFORM TRAP ---
    // Instantly move modals out of the main wrapper and directly into the body.
    (function relocateModals() {
        const qr = document.getElementById('qrModal');
        const note = document.getElementById('noteModal');
        if (qr && qr.parentNode !== document.body) document.body.appendChild(qr);
        if (note && note.parentNode !== document.body) document.body.appendChild(note);
    })();

    // --- 2. ROUTING ---
    function setMode(mode) { window.location.href = '?page=services&mode=' + mode; }
    
    // --- 3. SAFE SCROLL LOCK ---
    function lockAppScroll() { document.body.style.overflow = 'hidden'; }
    function unlockAppScroll() { document.body.style.overflow = ''; }

    // --- 4. MODAL CONTROLS ---
    function openAppModal(id, payload = null) {
        lockAppScroll();
        const modal = document.getElementById(id);
        
        if (id === 'qrModal' && payload) {
            const box = document.getElementById('qrPlace');
            box.innerHTML = '';
            new QRCode(box, { 
                text: payload, width: 200, height: 200,
                colorDark : "#1D1D1F", colorLight : "#ffffff", correctLevel : QRCode.CorrectLevel.L
            });
        } else if (id === 'noteModal' && payload) {
            document.getElementById('noteId').value = payload.id;
            document.getElementById('noteText').value = payload.txt;
        }

        modal.classList.add('active');
    }

    function closeAppModal(id) {
        document.getElementById(id).classList.remove('active');
        unlockAppScroll();
    }

    // --- 5. TOAST COPY ---
    function copyTxt(text) {
        navigator.clipboard.writeText(text).then(() => {
            if(navigator.vibrate) navigator.vibrate(20);
            const toast = document.getElementById('copyToast');
            toast.classList.add('show');
            setTimeout(() => toast.classList.remove('show'), 2500);
        });
    }
</script>
